## 请求体

<ParamField body="phone" type="string" required>
  手机号
  示例：`13800138000`
</ParamField>

<ParamField body="code" type="string" required>
  验证码
  示例：`123456`
</ParamField>

<ParamField body="area_code" type="string">
  Area 验证码
  示例：`+86`
</ParamField>

<ParamField body="product" type="string">
  产品代码
  示例：`R8888`
</ParamField>

</ResponseField>

<ResponseField name="message" type="string">
